\chapter{Informed Consent Form}\label{app:icf}
\noindent This appendix contains the Informed Consent Form (ICF) of the study (version 2/23Nov2026). The same form is used in three ways:
\begin{itemize}[leftmargin=*]
  \item as the first sections of the online preliminary survey, where respondents tick a box to consent;
  \item as the first sections of the online feedback form for beta testers and User Acceptance Testing (UAT) participants, with a tick box; and
  \item as the printed form, with Part II signed at the onboarding session, by the participants selected for the UAT.
\end{itemize}
\attachpdf{pdfs/AppA_ICF.pdf}{Informed Consent Form}

\chapter{Research Instruments}\label{app:instruments}
\noindent This appendix describes the instruments of the study and reproduces their items:
\begin{itemize}[leftmargin=*]
  \item the preliminary survey (researcher-developed);
  \item the test cases, task results (researcher-developed);
  \item the System Usability Scale (adopted from Brooke, 1996, with the product name substituted);
  \item the experience statements (researcher-developed); and
  \item the open-ended and follow-up questions (researcher-developed).
\end{itemize}
